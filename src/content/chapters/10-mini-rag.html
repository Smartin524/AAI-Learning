PYTHON REFERENCE · 10

数学、随机、日期与正则

dt 表示 datetime 对象。

括号内为传入参数；name=value 为关键字传参，*args / **kwargs 为多个位置 / 关键字参数。

导入方式

import math
import random
import statistics
import re
import collections
from datetime import datetime, timedelta

数学与统计

# 平方根
math.sqrt(x)

# 向上取整
math.ceil(x)

# 向下取整
math.floor(x)

# 对数；省略 base 时为自然对数
math.log(x, base)

# 以 10 为底的对数
math.log10(x)

# e 的 x 次方
math.exp(x)

# 正弦，输入弧度
math.sin(x)
# 余弦，输入弧度
math.cos(x)

# 角度转弧度
math.radians(degrees)

# 判断数值是否接近
math.isclose(a, b, rel_tol=1e-9, abs_tol=0.0)

# 均值
statistics.mean(data)

# 中位数
statistics.median(data)

# 样本标准差
statistics.stdev(data)

随机与计数

# 设置随机种子
random.seed(a=None)

# 返回 [0, 1) 随机浮点数
random.random()

# 随机整数，包含两端
random.randint(a, b)

# 从整数范围抽样，不含 stop
random.randrange(start, stop, step=1)

# 在两端之间抽取浮点数
random.uniform(a, b)

# 随机取一项
random.choice(sequence)

# 有放回抽样
random.choices(population, weights=None, k=1)
# 无放回抽样
random.sample(population, k)

# 原地打乱列表
random.shuffle(items)

# 创建计数器
collections.Counter(iterable)

# 按频数取前 n 项
counter.most_common(n=None)

# 创建带默认值工厂的字典
collections.defaultdict(default_factory)

# 创建双端队列
collections.deque(iterable=(), maxlen=None)

日期与时间

# 当前日期时间
datetime.now(tz=None)

# 创建日期时间
datetime(year, month, day, hour=0, minute=0, second=0)

# 按格式解析日期字符串
datetime.strptime(text, format)

# 按格式输出日期字符串
dt.strftime(format)

# 解析 ISO 日期时间
datetime.fromisoformat(text)
# 转 ISO 日期时间文本
dt.isoformat(sep="T", timespec="auto")

# 时间戳转日期时间
datetime.fromtimestamp(timestamp, tz=None)

# 日期时间转时间戳
dt.timestamp()

# 创建时间间隔
timedelta(days=0, seconds=0, hours=0, minutes=0)

正则表达式

# 查找首个匹配，返回 Match 或 None
re.search(pattern, string, flags=0)

# 从字符串开头匹配
re.match(pattern, string, flags=0)

# 匹配整个字符串
re.fullmatch(pattern, string, flags=0)

# 返回全部匹配结果
re.findall(pattern, string, flags=0)

# 返回 Match 迭代器
re.finditer(pattern, string, flags=0)
# 替换匹配内容
re.sub(pattern, repl, string, count=0, flags=0)

# 按正则拆分
re.split(pattern, string, maxsplit=0, flags=0)

# 编译可复用的正则对象
re.compile(pattern, flags=0)

# 取得匹配内容；无参数时为完整匹配
match.group(*groups)

# 转义正则特殊字符
re.escape(pattern)
