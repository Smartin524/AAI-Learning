PYTHON REFERENCE · 01

内置函数与类型转换

常用调用格式；用途写在代码注释中。

括号内为传入参数；name=value 为关键字传参，*args / **kwargs 为多个位置 / 关键字参数。

类型转换

# 转整数
int(x)

# 按进制解析整数文本
int(text, base=10)

# 转浮点数
float(x)

# 转字符串
str(obj)

# 转布尔值
bool(x)

# 转列表；list() 创建空列表
list(iterable)
# 转元组；tuple() 创建空元组
tuple(iterable)

# 转集合；set() 创建空集合
set(iterable)

# 转字典；dict() 创建空字典
dict(mapping_or_pairs)

# 文本转字节
bytes(text, encoding="utf-8")

# 创建复数
complex(real, imag=0)

数字与进制

# 绝对值
abs(x)

# 四舍六入五成双，保留指定位数
round(number, ndigits=None)

# 乘方
pow(base, exp)

# 模幂
pow(base, exp, mod)

# 返回商和余数
divmod(a, b)
# 二进制字符串
bin(n)

# 八进制字符串
oct(n)

# 十六进制字符串
hex(n)

# 字符转 Unicode 码点
ord(char)

# 码点转字符
chr(codepoint)

类型与对象查询

# 查询类型
type(obj)

# 判断对象类型
isinstance(obj, type_or_tuple)

# 判断子类关系
issubclass(cls, class_or_tuple)

# 判断能否调用
callable(obj)

# 读取属性，不存在时返回 default
getattr(obj, name, default)

# 设置属性
setattr(obj, name, value)
# 判断属性是否存在
hasattr(obj, name)

# 删除属性
delattr(obj, name)

# 列出属性名
dir(obj)

# 取得对象的属性字典
vars(obj)

# 取得对象标识
id(obj)

# 取得可哈希对象的哈希值
hash(obj)

输入、输出与帮助

# 打印多个对象
print(*objects, sep=" ", end="\n", file=None, flush=False)

# 读取一行文本
input(prompt="")

# 取得调试用字符串
repr(obj)
# 按格式说明格式化值
format(value, format_spec="")

# 查看帮助
help(obj)

# 进入调试器
breakpoint()
