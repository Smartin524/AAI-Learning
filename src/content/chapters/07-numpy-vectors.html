PYTHON REFERENCE · 07

NumPy 常用方法

a、b 表示 NumPy 数组。

括号内为传入参数；name=value 为关键字传参，*args / **kwargs 为多个位置 / 关键字参数。

导入方式

import numpy as np

创建数组

# 从数据创建数组
np.array(data, dtype=None)

# 转换为数组，尽可能复用已有数据
np.asarray(data, dtype=None)

# 全零数组
np.zeros(shape, dtype=float)

# 全一数组
np.ones(shape, dtype=float)
# 按指定值填充
np.full(shape, fill_value, dtype=None)

# 等步长序列，不包含 stop
np.arange(start, stop, step=1)

# 等间隔取 num 个点
np.linspace(start, stop, num=50, endpoint=True)

# 创建对角阵
np.eye(N, M=None, k=0)

形状与组合

# 改变形状；可用一个 -1 自动推断维度
a.reshape(shape)

# 展平，尽可能返回视图
a.ravel()

# 展平并复制
a.flatten()

# 交换维度
np.transpose(a, axes=None)

# 插入长度为 1 的维度
np.expand_dims(a, axis)
# 移除长度为 1 的维度
np.squeeze(a, axis=None)

# 沿已有轴拼接
np.concatenate(arrays, axis=0)

# 沿新轴堆叠
np.stack(arrays, axis=0)

# 按等份数或索引切分
np.split(a, indices_or_sections, axis=0)

# 转换元素类型
a.astype(dtype)

计算与筛选

# 求和
np.sum(a, axis=None, keepdims=False)

# 均值
np.mean(a, axis=None, keepdims=False)

# 标准差
np.std(a, axis=None, ddof=0)

# 最小值
np.min(a, axis=None)

# 最大值
np.max(a, axis=None)

# 最大值索引
np.argmax(a, axis=None)

# 最小值索引
np.argmin(a, axis=None)

# 按条件选择值
np.where(condition, x, y)
# 限制数值范围
np.clip(a, a_min, a_max)

# 去重，可同时返回计数
np.unique(a, return_counts=False)

# 排序后的数组
np.sort(a, axis=-1)

# 排序索引
np.argsort(a, axis=-1)

# 逐项判断 NaN
np.isnan(a)

# 点积；二维输入为矩阵乘法
np.dot(a, b)

# 矩阵乘法
np.matmul(a, b)

# 向量或矩阵范数
np.linalg.norm(a, ord=None, axis=None)

随机数与保存

# 创建随机数生成器 rng
np.random.default_rng(seed=None)

# 生成 [0, 1) 均匀随机数
rng.random(size=None)

# 生成整数，上界不包含
rng.integers(low, high=None, size=None)

# 正态分布采样
rng.normal(loc=0.0, scale=1.0, size=None)

# 按概率抽样
rng.choice(a, size=None, replace=True, p=None)
# 原地随机打乱
rng.shuffle(a, axis=0)

# 返回打乱后的副本
rng.permutation(a, axis=0)

# 保存 .npy 数组
np.save(file, a)

# 读取数组文件
np.load(file, allow_pickle=False)
