MySQL 常用命令

SQL 是查询语言，MySQL 是使用 SQL 的关系型数据库软件。以下补充 MySQL 8+ 的连接、管理和工具命令；查询语法见上一页。

连接与命令行

# 在终端执行；-p 后交互输入密码
mysql -u user_name -p

# 指定主机、端口和数据库
mysql -h 127.0.0.1 -P 3306 -u user_name -p app_db

# 强制使用 TCP 连接
mysql --protocol=TCP -h 127.0.0.1 -u user_name -p

# 查看客户端版本
mysql --version
# 在终端执行一条 SQL 后退出
mysql -u user_name -p -e "SELECT VERSION();"

# 以 UTF-8 连接指定数据库
mysql --default-character-set=utf8mb4 -u user_name -p app_db

# 以下在 mysql> 提示符内执行
status

# 取消尚未提交的输入
\c

# 退出客户端
exit

数据库与表结构

-- 查看数据库 / 当前数据库 / 服务端版本
SHOW DATABASES;
SELECT DATABASE();
SELECT VERSION();

-- 创建数据库并指定字符集
CREATE DATABASE IF NOT EXISTS app_db
CHARACTER SET utf8mb4;

-- 切换数据库
USE app_db;

-- 当前数据库中的表
SHOW TABLES;
SHOW FULL TABLES;
-- 列、索引及完整建表语句
DESCRIBE users;
SHOW COLUMNS FROM users;
SHOW INDEX FROM users;
SHOW CREATE TABLE users;

-- 查看表的引擎等状态信息
SHOW TABLE STATUS LIKE 'users';

-- 查看会话字符集设置
SHOW VARIABLES LIKE 'character_set%';

-- 删除整个数据库及其数据
DROP DATABASE IF EXISTS demo_db;

MySQL 常用写法

-- 自增主键与自动时间戳
CREATE TABLE items (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  name VARCHAR(100) NOT NULL,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 最近一次生成的自增编号（当前连接）
SELECT LAST_INSERT_ID();

-- 修改列类型；按需重写默认值等完整定义
ALTER TABLE items
MODIFY COLUMN name VARCHAR(200) NOT NULL;

-- 重命名表
RENAME TABLE items TO inventory;
-- 空值替代与条件表达式
SELECT IFNULL(nickname, name) FROM users;
SELECT IF(score >= 60, 'pass', 'fail') FROM users;

-- 空值安全的相等比较
SELECT NULL <=> NULL;

-- 分组拼接；长度受 group_concat_max_len 限制
SELECT city,
  GROUP_CONCAT(name ORDER BY name SEPARATOR ', ')
FROM users GROUP BY city;

-- 主键冲突时更新（示例值）
INSERT INTO users (id, name)
VALUES (1, 'Alice')
ON DUPLICATE KEY UPDATE name = 'Alice';

-- 最近一条语句影响的行数
SELECT ROW_COUNT();

导入与导出（终端）

# 导出一个数据库；适用于 InnoDB 表的一致性快照
# 导出期间避免 ALTER / DROP 等结构变更
mysqldump -u user_name -p --single-transaction app_db > app_db.sql

# 仅导出表结构
mysqldump -u user_name -p --no-data app_db > schema.sql

# 仅导出指定表
mysqldump -u user_name -p app_db users orders > tables.sql

# 导出并包含建库、切库语句
mysqldump -u user_name -p --databases app_db > database.sql
# 恢复到已存在的数据库；会执行文件中的 SQL
mysql -u user_name -p app_db < app_db.sql

# 恢复包含 CREATE DATABASE / USE 的备份
mysql -u user_name -p < database.sql

# 在 mysql> 提示符内执行 SQL 文件
source /path/to/script.sql

# 在终端把查询结果导出为制表符分隔文本
mysql -u user_name -p --batch app_db -e "SELECT id, name FROM users" > users.tsv

用户、会话与排查

-- 连接用户与实际匹配的授权账户
SELECT USER(), CURRENT_USER();

-- 查看当前账户权限
SHOW GRANTS;

-- 创建本机账户（需要管理权限）
CREATE USER 'app_user'@'localhost'
IDENTIFIED BY 'replace_with_password';

-- 授予指定数据库的读写权限
GRANT SELECT, INSERT, UPDATE, DELETE
ON app_db.* TO 'app_user'@'localhost';

-- 撤销删除权限
REVOKE DELETE ON app_db.*
FROM 'app_user'@'localhost';
-- 查看连接；他人的连接需额外权限
SHOW FULL PROCESSLIST;
SELECT CONNECTION_ID();

-- 查看上一条语句的警告
SHOW WARNINGS;

-- 查看当前会话事务设置
SELECT @@autocommit, @@transaction_isolation;

-- 查看服务状态计数
SHOW GLOBAL STATUS LIKE 'Threads_connected';
SHOW GLOBAL STATUS LIKE 'Slow_queries';

-- 查看查询执行计划，不执行该 SELECT
EXPLAIN SELECT * FROM users WHERE id = 1;
