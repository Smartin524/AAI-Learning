PYTHON REFERENCE · 02

容器：list、dict、set、tuple

常用调用格式；用途写在代码注释中。

括号内为传入参数；name=value 为关键字传参，*args / **kwargs 为多个位置 / 关键字参数。以下代码块为独立调用速查，变量需替换为你的数据，不应整块顺序运行；默认值与示例值以注释为准。

列表 list

# 末尾追加一个元素
items.append(value)

# 末尾追加多个元素
items.extend(iterable)

# 在索引处插入
items.insert(index, value)

# 删除首个匹配值
items.remove(value)

# 删除并返回指定元素
items.pop(index=-1)

# 在指定范围查找首个索引；start、stop 可省略
items.index(value, start, stop)
# 统计出现次数
items.count(value)

# 原地排序，返回 None
items.sort(key=None, reverse=False)

# 原地反转，返回 None
items.reverse()

# 浅复制
items.copy()

# 清空列表
items.clear()

字典 dict

# 读取键，不存在时返回默认值
d.get(key, default=None)

# 取得键视图
d.keys()

# 取得值视图
d.values()

# 取得键值对视图
d.items()

# 合并键值对，原地更新
d.update(other, **kwargs)

# 键不存在时写入默认值，返回该键的值
d.setdefault(key, default=None)
# 删除并返回值；可省略 default，缺键时报错
d.pop(key, default)

# 删除并返回最后插入的键值对
d.popitem()

# 浅复制
d.copy()

# 清空字典
d.clear()

# 用多个键创建字典，共享同一个 value
dict.fromkeys(iterable, value=None)

集合 set

# 添加一个元素
s.add(value)

# 添加多个集合或可迭代对象的元素
s.update(*iterables)

# 删除元素，不存在时报错
s.remove(value)

# 删除元素，不存在时忽略
s.discard(value)

# 删除并返回任意一个元素
s.pop()

# 返回并集
s.union(*others)

# 返回交集
s.intersection(*others)

# 返回差集
s.difference(*others)

# 返回对称差集
s.symmetric_difference(other)
# 原地保留交集
s.intersection_update(*others)

# 原地删除其他集合中的元素
s.difference_update(*others)

# 原地改为对称差集
s.symmetric_difference_update(other)

# 是否为子集
s.issubset(other)

# 是否为超集
s.issuperset(other)

# 是否没有共同元素
s.isdisjoint(other)

# 浅复制
s.copy()

# 清空集合
s.clear()

元组与序列

# 统计元组中的出现次数
t.count(value)

# 查找元组中的首个索引；start、stop 可省略
t.index(value, start, stop)
# 取得元素数量
len(sequence)

# 切片；边界不包含 stop，三项均可省略
sequence[start:stop:step]
