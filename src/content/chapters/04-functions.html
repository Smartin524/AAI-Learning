PYTHON REFERENCE · 04

遍历、排序与函数工具

内置遍历函数无需导入；下面三个模块用于标准库工具。

括号内为传入参数；name=value 为关键字传参，*args / **kwargs 为多个位置 / 关键字参数。

导入方式

import itertools
import functools
import operator

遍历

# 生成从 0 到 stop 前的整数
range(stop)

# 按步长生成整数
range(start, stop, step=1)

# 同时取得编号和元素
enumerate(iterable, start=0)

# 并行配对，长度以最短输入为准
zip(*iterables)
# 创建迭代器
iter(iterable)

# 取得下一项；省略 default 时耗尽会报错
next(iterator, default)

# 返回反向迭代器
reversed(sequence)

排序与汇总

# 返回排序后的新列表
sorted(iterable, key=None, reverse=False)

# 取得最小项
min(iterable, key=None)

# 取得多个参数中的最小项
min(a, b, *others, key=None)

# 取得最大项
max(iterable, key=None)

# 取得多个参数中的最大项
max(a, b, *others, key=None)
# 求和
sum(iterable, start=0)

# 是否全部为真
all(iterable)

# 是否至少一个为真
any(iterable)

# 逐项调用函数，返回迭代器
map(func, *iterables)

# 保留判断为真的元素；func=None 时按元素真假筛选
filter(func, iterable)

itertools

# 依次连接多个可迭代对象
itertools.chain(*iterables)

# 按范围截取迭代器
itertools.islice(iterable, start, stop, step=1)

# 笛卡尔积
itertools.product(*iterables, repeat=1)
# 取 r 项组合
itertools.combinations(iterable, r)

# 取 r 项排列
itertools.permutations(iterable, r=None)

# 累计计算；默认累计求和
itertools.accumulate(iterable, func=operator.add)

functools 与 operator

# 累计归约；可省略 initial
functools.reduce(func, iterable, initial)

# 预先固定部分参数
functools.partial(func, *args, **kwargs)

# 缓存函数结果的装饰器
functools.lru_cache(maxsize=128, typed=False)
# 按索引或键生成取值函数
operator.itemgetter(*items)

# 按属性名生成取值函数
operator.attrgetter(*attrs)
