Pandas 常用方法

常用调用格式；用途写在代码注释中。import pandas as pd；df 为 DataFrame，s 为 Series。以下展示常用参数组合。

括号内为传入参数；name=value 为关键字传参，*args / **kwargs 为多个位置 / 关键字参数。以下代码块为独立调用速查，变量需替换为你的数据，不应整块顺序运行；默认值与示例值以注释为准。

创建、读取与查看

# 创建表格
pd.DataFrame(data, index=None, columns=None)

# 创建一列数据
pd.Series(data, index=None, name=None)

# 读取 CSV
pd.read_csv(filepath, sep=",", usecols=None, dtype=None)

# 读取 Excel
pd.read_excel(io, sheet_name=0, usecols=None)

# 查看前 n 行
df.head(n=5)
# 查看后 n 行
df.tail(n=5)

# 查看列类型及非空数量
df.info()

# 汇总数值列统计
df.describe()

# 保存 CSV
df.to_csv(path, index=False)

# 保存 Excel
df.to_excel(path, index=False)

选择与清理

# 按标签选择；这是索引器
df.loc[row_labels, column_labels]

# 按位置选择；这是索引器
df.iloc[row_positions, column_positions]

# 按表达式筛选行
df.query(expr)

# 删除指定列
df.drop(columns=names)

# 重命名列
df.rename(columns=mapping)

# 转换类型或按列类型映射转换
df.astype(dtype)
# 判断缺失值
df.isna()

# 填充缺失值
df.fillna(value)

# 删除有缺失值的行或列
df.dropna(axis=0, how="any", subset=None)

# 删除重复行
df.drop_duplicates(subset=None, keep="first")

# 转换为数值
pd.to_numeric(arg, errors="raise")

# 转换为日期时间
pd.to_datetime(arg, errors="raise", format=None)

计算、分组与排序

# 按列值排序
df.sort_values(by, ascending=True)

# 按索引排序
df.sort_index(axis=0, ascending=True)

# 创建分组对象
df.groupby(by, dropna=True)

# 汇总分组；func 可为函数名、列表或字典
grouped.agg(func)

# 按列或行汇总
df.agg(func, axis=0)

# 按列或行应用函数
df.apply(func, axis=0)
# 按映射或函数转换每个值
s.map(arg)

# 统计各值频数或比例
s.value_counts(normalize=False, dropna=True)

# 取得唯一值
s.unique()

# 统计唯一值数量
s.nunique(dropna=True)

# 随机抽取行
df.sample(n=1, random_state=None)

合并与重塑

# 按行或列拼接
pd.concat(objs, axis=0, ignore_index=False)

# 按键连接表格
pd.merge(left, right, how="inner", on=None)

# 连接表格，默认按索引
df.join(other, on=None, how="left")

# 长表转宽表，重复组合会报错
df.pivot(index=rows, columns=cols, values=values)
# 透视并汇总重复组合
df.pivot_table(values=values, index=rows, columns=cols, aggfunc="mean")

# 宽表转长表
df.melt(id_vars=None, value_vars=None, var_name=None, value_name="value")

# 将列设为索引
df.set_index(keys, drop=True)

# 重置索引
df.reset_index(drop=False)
